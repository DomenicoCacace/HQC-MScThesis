\subsection{Field theory}
\begin{definition}[Group]
    Let $G$ be a set. The structure $(G, \cdot)$ is a \emph{group} if it satisfies the following axioms:
    \begin{itemize}
        \item Closure: For all $a, b \in G$, $a \cdot b \in G$.
        \item Associativity: For all $a, b, c \in G$, $(a \cdot b) \cdot c = a \cdot (b \cdot c)$.
        \item Identity: There exists an element $e \in G$ such that for all $a \in G$, $e \cdot a = a \cdot e = a$.
        \item Inverse: For all $a \in G$, there exists an element $a^{-1} \in G$ such that $a \cdot a^{-1} = a^{-1} \cdot a = e$.
    \end{itemize}
    If the group also satisfies the commutative property (i.e. $a \cdot b = b \cdot a$ for all $a, b \in G$), then it is called an \emph{abelian group}.
    We usually call $(G, +)$ the \emph{additive} group and $(G, \cdot)$ the \emph{multiplicative} group, with identity elements denoted by $0$ and $1$, respectively.
\end{definition}

\begin{definition}[Ring]
    Let $R$ be a set. The structure $(R, +, \cdot)$ is a \emph{ring} if it satisfies the following properties:
    \begin{itemize}
        \item Additive group: $(R, +)$ is an abelian group.
        \item Associativity w.r.t. multiplication: For all $a, b, c \in R$, $(a \cdot b) \cdot c = a \cdot (b \cdot c)$.
        \item Distributivity: For all $a, b, c \in R$, $a \cdot (b + c) = a \cdot b + a \cdot c$ and $(a + b) \cdot c = a \cdot c + b \cdot c$.
    \end{itemize}
\end{definition}

\begin{definition}[Field]
    Let $K$ be a set. The structure $(K, +, \cdot)$ is a \emph{field} if it satisfies the following properties:
    \begin{itemize}
        \item Additive group: $(K, +)$ is an abelian group.
        \item Multiplicative group: $(K\setminus\{0\}, \cdot)$ is an abelian group.
        \item Distributivity: For all $a, b, c \in K$, $a \cdot (b + c) = a \cdot b + a \cdot c$ and $(a + b) \cdot c = a \cdot c + b \cdot c$.
    \end{itemize}
    Any field is also a ring, but the converse is not necessarily true.
\end{definition}

\begin{definition}[Characteristic]
    Let $K$ be a field. The \emph{characteristic} of $K$ is the smallest positive integer $n$ such that
    \begin{equation*}
        \underbrace{1 + \cdots + 1}_\text{$n$ times} = 0
    \end{equation*}
\end{definition}

\begin{definition}[Order]
    Let $K$ be a field. The \emph{order} of $K$ is the amount of elements in $K$; if the order of $K$ is finite, it is also called a Galois field.
    A Galois field of order $q$, denoted as $\mathds{F}_q$ or $GF(q)$, exists if and only if $q=p^n$ for some prime $p$; all Galois fields of order $q$ are isomorphic.
\end{definition}
Galois fields can be constructed in two ways:
\begin{itemize}
    \item Primitive fields: $GF(p) \cong \mathds{Z}_p$, with $p$ prime, is a field of order and characteristic $p$.
    \item $GF(p^n) = GF(p)[X]/P(X)$ with $P(X)$ irreducible polynomial in $GF(p)[X]$ of degree $n$,
        is a field of order $p^n$ and characteristic $p$.
\end{itemize}

\begin{definition}[Primitive element]
    Let $GF(q)$ be a Galois field. An element $a \in GF(q)$ is a \emph{primitive element} if every nonzero element of $GF(q)$ is a power of $a$;
    in other words, $a$ is a generator of the multiplicative group of $GF(q)$. Since the multiplicative group of $GF(q)$ has order $q-1$,
    primitive elements are also called $(q-1)$-th roots of unity.
\end{definition}

\begin{definition}[Field extension]
    Let $K$ and $L$ be fields. If $K \subseteq L$, then we say that $K$ is a \emph{subfield} of $L$; conversely, 
    we say that $L$ is a \emph{field extension} of $K$.
    Given a field extension, denoted as $L/K$, the field $L$ is a vector space over $K$; the dimension of the vector space
    defines the \emph{degree of the extension}, and is denoted by $[L:K]$. In the case of finite fields, 
    $[GF(p^n):GF(p)] = n$.
\end{definition}

\begin{example}[Binary fields]
    The smallest Galois field is $GF(2)$, which is isomorphic to $\mathds{Z}_2 = \{0,1\}$.
    We can build $\mathds{F}_8 = GF(2^3)$ as $GF(2)[X]/\langle X^3+X+1\rangle$.
    The elements of $\mathds{F}_8$ are then the binary polynomials of degree strictly less than $n=3$; to avoid confusion, 
    the elements are written as polynomials in the variable $\alpha$, where $P(\alpha) = 0$.\\
    We have that $\mathds{F}_8 = \{0, 1, \alpha, 1+\alpha, \alpha^2, 1+\alpha^2, \alpha + \alpha^2, 1+\alpha+\alpha^2\}$, so the order of the field is $8$.
    Moreover, since $1+1=0$ in $GF(2)$, we have that $1+1=0$ in $\mathds{F}_8$ as well, so the characteristic of $\mathds{F}_8$ is $2$.\\
    By construction, $\mathds{F}_8$ is a field extension of $GF(2)$ of degree $3$; the vector space $\mathds{F}_8$ is thus a vector space over $GF(2)$ of dimension $3$,
    having as basis the elements $\{1, \alpha, \alpha^2\}$.
\end{example}

\subsection{Frobenius automorphism}
    \begin{definition}[Frobenius map]
    Let $\mathds{F}_2$ be the binary field. The Frobenius map is the automorphism defined as:
    \begin{equation}
        \phi_2 : \mathds{F}_2 \rightarrow \mathds{F}_2, \quad \phi_2(x) = x^2
    \end{equation}
    \end{definition}
    Let $\Sigma$ be a set; with the notation $\phi_2(\Sigma)$ we denote the set built as $\{\phi_2(x) \mid x \in \Sigma\}$;
    moreover, with the notation $\phi_2^{\circ j}$ we denote the function applying the Frobenius map $j$ times.

    \begin{definition}[Order of $\phi_2$]
        Let $\Sigma$ be a set. The order of $\phi_2$ on $\Sigma$, denoted as $Ord_{\phi_2}(\Sigma)$ is the smallest positive integer $j$ such that $\phi_2^{\circ j}(\Sigma) = \Sigma$.
    \end{definition}

    \begin{definition}[Frobenius partition]
        Given the sets $\Sigma, \Omega$ such that $Ord_{\phi_2}(\Sigma)=j$, we say that $\Sigma$ is a Frobenius partition
        of $\Omega$ if
        \begin{equation}
            \Omega = \Sigma \cup \phi_2(\Sigma) \cup \cdots \cup \phi_2^{\circ (j-1)}(\Sigma)
        \end{equation}
        and all the sets $\phi_2^{\circ i}(\Sigma)$ are pairwise disjoint.
    \end{definition}

\subsection{Notation}
Throughout this document, we will use the following algebraic notation:
\begin{itemize}
    \item $\mathds{Z}$: the ring of integers.
    \item $\mathds{F}_2$: the binary Galois field $GF(2)$.
    \item $\mathds{F}_{2^n}$: the Galois field $GF(2^n)$.
    \item $\mathcal{V} = \mathds{F}_2^n$: the vector space of dimension $n$ over $GF(2)$, with canonical basis $\{1, \alpha, \cdots, \alpha^{n-1}\}$.
    \item $\mathcal{R} = \mathds{F}_2[X]/\langle X^n-1\rangle$: the ring of binary polynomials of degree less than $n$.
    \item $\texttt{wt}(\cdot)$: the Hamming weight of a vector \textit{i.e.} the number of its non-zero coordinates.
\end{itemize}
