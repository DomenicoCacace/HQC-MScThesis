            height=70mm, width=90mm,

            bar width=0.2,
            ybar=1pt,
            enlarge x limits={abs=0.5},

            %yticklabel=\pgfmathprintnumber{\tick},

            % no ticks on y axis

            yticklabels={},
            %xlabel={Wait States},
            xtick=data,
            xticklabels from table={#1}{Label},
        %
            %nodes near coords,
            %nodes near coords style={font=\tiny,
            %                        /pgf/number format/.cd,
            %                                    precision=0,
            %                                    zerofill,
            %                        },

                        cells={anchor=west},
                        font=\footnotesize,
                        }
