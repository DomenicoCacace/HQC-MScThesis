\newcommand{\ARMtimingplot}[1]{
    \begin{tikzpicture}
        \centering
        \begin{axis}[
            height=65mm, width=90mm,
            bar width=0.2,
            ybar=0pt,
            enlarge x limits={abs=0.5},
            ymode=log,
            ytick={},
            yticklabels={},
            xtick=data,
            xticklabels from table={#1}{Label},
        ]
        \addplot[preaction={fill=blue!30}, pattern=north west lines, pattern color=blue!50!black] table [x expr=\coordindex, y=reference (R3)]{#1};
        \addplot[preaction={fill=red!30}, pattern=crosshatch dots, pattern color=purple!70] table [x expr=\coordindex, y=reference (R4)]{#1};
        \addplot[preaction={fill=yellow!30}, pattern=north east lines, pattern color=orange!70] table [x expr=\coordindex, y=sparsemul (R4)]{#1};
        \addplot[preaction={fill=green!30}, pattern=crosshatch, pattern color=green!40!black] table [x expr=\coordindex, y=fastmul]{#1};
        \end{axis}
    \end{tikzpicture}
}
