\newcommand{\ARMtimingplot}[1]{
    \begin{tikzpicture}
        \centering
        \begin{axis}[
            height=70mm, width=90mm,
            bar width=0.2,
            ybar=0pt,
            enlarge x limits={abs=0.5},
            ymode=log,
            ytick={},
            yticklabels={},
            xtick=data,
            xticklabels from table={#1}{Label},
        ]
        \addplot[fill=blue!30] table [x expr=\coordindex, y=reference (R3)]{#1};
        \addplot[fill=red!30] table [x expr=\coordindex, y=reference (R4)]{#1};
        \addplot[fill=orange!30] table [x expr=\coordindex, y=sparsemul (R4)]{#1};
        \addplot[fill=green!30] table [x expr=\coordindex, y=fastmul]{#1}; 
        \end{axis}
    \end{tikzpicture}
}
