\begin{frame}{Classic computers \textit{vs} classic cryptography}
    \uncover<1->{\begin{block}{Symmetric cryptosystems}
        \textbf{Key enumeration}: perform an exhaustive search on the whole keyspace, $\mathcal{O}(n)$ time
    \end{block}}
    \uncover<2->{
    \begin{block}{Asymmetric cryptosystems}
        \textbf{General number field sieve}: factor a composite number $n$ in $\mathcal{O}(e^{1.9(\log{n})^{1/3})(\log{\log{n}})^{2/3}})$ (subexponential) time    \end{block}}
\end{frame}

\begin{frame}{Quantum computers \textit{vs} classic cryptography}
    \uncover<1->{\begin{block}{Symmetric cryptosystems}
        \textbf{Grover's algorithm}: search in a database of $n$ elements in $\mathcal{O}(n^{1/2})$ time and $\mathcal{O}(\log n)$ space
    \end{block}}
    \uncover<2->{\begin{block}{Asymmetric cryptosystems}
        \textbf{Shor's algorithm}: factor a composite number $n$, on a quantum computer, in $\mathcal{O}((\log{n})^2(\log{\log{n}})(\log{\log{\log{n}}}))$ (polynomial) time~\cite{DBLP:conf/focs/Shor94}
    \end{block}}
\end{frame}
