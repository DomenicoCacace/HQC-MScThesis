\section{Keypair generation}
\begin{algorithm}
\caption{Keygen}
\KwData{$p$: a prime such that $\mathrm{ord}_{p}(2)=p-1$\newline
        $\mathbf{R}$: the polynomial ring $\mathbb{Z}_2[x]/\langle x^p+1 \rangle$ \newline
        $\mathbf{R}_{w}$: the elements of the polynomial ring $\mathbb{Z}_2[x]/\langle x^p+1 \rangle$ with exactly $w$ non-null coefficients\newline      
        $\gets$ denotes uniform sampling from the r.h.s. \newline
        $\ssplit{a} = (a_0,\ldots,a_{n-1}), n > 0$ an additive share splitting of $a$ in $n$ shares over $\mathbf{R}$}
 \KwIn{none}
 \KwOut{$\kpub \in \mathbf{R}\times \mathbf{R}; \kpri \in \mathbf{R} \times \mathbf{R}$}
 $h \sample \mathbf{R}$\;
 $(x,y) \sample \mathbf{R}_w \times \mathbf{R}_w$\;
 $\ssplit{y} \gets (y+\sum_{i=0}^{n-2} \mathtt{rnd}_i, \mathtt{rnd}_0, \ldots \mathtt{rnd}_{n-2})$\;
 $\ssplit{s} \gets h \ssplit{y}$ \tcp*{share split by scalar mul acts as vector by scalar mul}
 $\ssplit{s} \gets x + \ssplit{s}$ \tcp*{share split plus scalar sum adds scalar to first share only }
 \KwRet $\kpri=(x,y)$, $\kpub = (h,s)$\;
\end{algorithm}

\section{Encrypt}
\begin{algorithm}
\caption{Encrypt($\kpub$, $m$)}
\KwData{$p$: a prime such that $\mathrm{ord}_{p}(2)=p-1$\newline
        $\mathbf{R}$: the polynomial ring $\mathbb{Z}_2[x]/\langle x^p+1 \rangle$ \newline
        $\mathbf{R}_{w}$: the elements of the polynomial ring $\mathbb{Z}_2[x]/\langle x^p+1 \rangle$ with exactly $w$ non-null coefficients\newline      
        $\gets$ denotes uniform sampling from the r.h.s. \newline
        $\ssplit{a} = (a_0,\ldots,a_{n-1}), n > 0$ an additive share splitting of $a$ in $n$ shares over $\mathbf{R}$\newline
        $G$: a $\mathbb{F}_2^{k \times p}$ matrix generating an efficiently correctable code}
 \KwIn{$\kpub$, $m\in \{0,1\}^k$}
 \KwOut{$\kpub \in \mathbf{R}\times \mathbf{R}; \kpri \in \mathbf{R} \times \mathbf{R}$}
 $e \sample \mathbf{R}_{w_e}$\;
 $(r_1,r_2) \sample \mathbf{R}_{w_r} \times \mathbf{R}_{w_r}$\;
 $\ssplit{r_1} \gets (r_1+\sum_{i=0}^{n-2} \mathtt{rnd}_i, \mathtt{rnd}_0, \ldots \mathtt{rnd}_{n-2})$\;
 $\ssplit{r_2} \gets (r_2+\sum_{i=0}^{n-2} \mathtt{rnd}_i, \mathtt{rnd}_0, \ldots \mathtt{rnd}_{n-2})$\;
 $\ssplit{u} \gets \ssplit{r_1} + h\ssplit{r_2}$\;
 $\ssplit{m} \sample (\{0,1\}^k)^n $\;
 $\ssplit{v} \gets \ssplit{m}G + s \ssplit{r_2} + e $\;\tcp{computed from left to right, or share-split $e$}
 \KwRet $(u,v)$\;
\end{algorithm}


\section{Decrypt}
\begin{algorithm}
\caption{Decrypt($\kpri$, $c$)}
\KwData{$p$: a prime such that $\mathrm{ord}_{p}(2)=p-1$\newline
        $\mathbf{R}$: the polynomial ring $\mathbb{Z}_2[x]/\langle x^p+1 \rangle$ \newline
        $\mathbf{R}_{w}$: the elements of the polynomial ring $\mathbb{Z}_2[x]/\langle x^p+1 \rangle$ with exactly $w$ non-null coefficients\newline      
        $\gets$ denotes uniform sampling from the r.h.s. \newline
        $\ssplit{a} = (a_0,\ldots,a_{n-1}), n > 0$ an additive share splitting of $a$ in $n$ shares over $\mathbf{R}$\newline
        $G$: a $\mathbb{F}_2^{k \times p}$ matrix generating an efficiently correctable code}
 \KwIn{$\kpri=(x,y) \in \mathbf{R}_w \times \mathbf{R}_w$, $c = (u,v) \in \mathbf{R} \times \mathbf{R}$}
 \KwOut{$m\in \{0,1\}^k$}
 $\ssplit{y} \gets (y+\sum_{i=0}^{n-2} \mathtt{rnd}_i, \mathtt{rnd}_0, \ldots \mathtt{rnd}_{n-2})$\;
 $\ssplit{uy} \gets u \ssplit{y}$\;
 $\ssplit{\bar{m}} \sample (\{0,1\}^k)^n$ \tcp*{random, share split information word}
 $\ssplit{\bar{c}} \gets \ssplit{\bar{m}}G$ \tcp*{random, share split code word}
 $a \gets \text{\sc Collapse}(v ,\ssplit{\bar{c}}) $ \tcp*{add all shares, in order, to $v$}
 $m \gets \text{\sc Decode}(\text{\sc Collapse}(a,\ssplit{uv}))$\;
 \KwRet $\text{\sc Collapse}(m ,\ssplit{\bar{m}})$\;
\end{algorithm}
