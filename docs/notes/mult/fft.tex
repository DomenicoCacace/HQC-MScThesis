


\subsubsection{Cooley-Tukey FFT}
The Cooley-Tukey FFT~\cite{cooley1965algorithm} is the most common FFT algorithm. It is a \textit{divide et impera} algorithm, which recursively expresses a DFT of
        size $N = N_1 N_2$ into $N_1$ smaller DFTs of size $N_2$ each. The simplest form of the Cooley-Tukey algorithm is the radix-2 decimation-in-time FFT, which computes the DFTs of
        the even and odd indexed inputs separately and then combines the results. Assuming that N is a power of 2, we can rewrite the DFT as:
        \begin{align*}
            X_k & = \sum_{n=0}^{N-1} x_n e^{-\frac{2\pi i}{N}nk} \\
                & = \sum_{m=0}^{N/2-1}x_{2m}e^{-\frac{2\pi i}{N}(2m)k} + \sum_{m=0}^{N/2-1}x_{2m+1}e^{-\frac{2\pi i}{N}(2m+1)k} \\
                & = \sum_{m=0}^{N/2-1}x_{2m}e^{-\frac{2\pi i}{N/2}mk} + e^{-\frac{2\pi i}{N}k}\sum_{m=0}^{N/2-1}x_{2m+1}e^{-\frac{2\pi i}{N/2}mk} \\
                & = E_k + e^{-\frac{2\pi i}{N}k}O_k \qquad \text{ for } k = 0, \cdots, N/2-1
        \end{align*}
        where we split the DFT into two smaller DFTs, on the even and odd indexes respectively. Because of the periodicity of the complex exponential, we can also write:
        \begin{align*}
            X_{k+N/2} & = \sum_{m=0}^{N/2-1}x_{2m}e^{-\frac{2\pi i}{N/2}m(k+N/2)} + e^{-\frac{2\pi i}{N}(k+N/2)}\sum_{m=0}^{N/2-1}x_{2m+1}e^{-\frac{2\pi i}{N/2}m(k+N/2)} \\
                & = \sum_{m=0}^{N/2-1}x_{2m}e^{-\frac{2\pi i}{N/2}mk}e^{-2\pi mi} + e^{-\frac{2\pi i}{N}k}e^{-\pi i}\sum_{m=0}^{N/2-1}x_{2m+1}e^{-\frac{2\pi i}{N/2}mk}e^{-2\pi mi} \\
                & = \sum_{m=0}^{N/2-1}x_{2m}e^{-\frac{2\pi i}{N/2}mk} - e^{-\frac{2\pi i}{N}k}\sum_{m=0}^{N/2-1}x_{2m+1}e^{-\frac{2\pi i}{N/2}mk} \\
                & = E_k - e^{-\frac{2\pi i}{N}k}O_k \qquad \text{ for } k = 0, \cdots, N/2-1
        \end{align*}
        This means that we can express a DFT of size $N$ as two DFTs of size $N/2$, where we can re-use intermediate results to calculate multiple outputs. 
        Since the base case of the recursion has complexity $\mathcal{O}(n)$ we get the recurrence relation $T(n) = 2T(n/2) + \mathcal{O}(n)$;
        following the Master theorem~\cite{bentley1980general}, the computational complexity of the Cooley-Tukey FFT is $\mathcal{O}(n\log n)$.



\subsubsection{Cantor basis}
In~\cite{cantor1989arithmetical} Cantor showed how to get a special basis in which it is possible to compute the FFT efficiently
    for any $m = p^n$, with $p$ a prime number; later in~\cite{gao2010additive} Gao and Mateer used a simpler construction for the special case
    $p=2$, where the Cantor basis $\mathbf{v} = (v_0, \cdots, v_{m-1})$ follows the structure:
    \begin{equation}
        v_0 = 1, \quad v_i^2+v_i = v_i-1 \text{ for } i \geq 1
    \end{equation}
    With respect to the Cantor basis $\mathbf{v}$, we define its sequence of subspaces as:
    \begin{equation}
        W_0 := \{0\}, \quad W_i := span\{v_0, \cdots, v_{i-1}\} \text{ for } i \geq 1
    \end{equation}

    By construction, we have that d$W_0 \subset W_1 \subset \cdots \subset W_{m-1} = \mathds{F}_{2^m}$ 
    and the basis for a smaller space is contained into the basis of a larger space: this means that 
    we can arbitrarily treat elements of smaller fields as elements in the larger fields by padding zeros to the extra dimensions.

    With respect to each element of the basis $v_i$ we can define its subspace polynomial $s_i(x)$ as:
    \begin{equation}
        s_i(x) = \prod_{a \in W_i} (x - a)
    \end{equation}
    Since $\dim(W_i) = i$ we have $\deg(s_i(x)) = |V_i| = 2^i$.
    Such polynomials, as shown in~\cite{cantor1989arithmetical, gao2010additive}, satisfy the following useful properties:
    \begin{itemize}
        \item linearity: $s_i(x+y) = s_i(x) + s_i(y)$
        \item recursivity: $s_{i+j}(x) = s_i(s_j(x))$
    \end{itemize}

\subsubsection{Segmentation of large polynomials}
Suppose we are multiplying two binary polynomials $f(x)$ and $g(x) \in \mathds{F}_2[x]$, represented as sequences of $d$ bits:
    \begin{equation*}
        a(x) = a_0 + a_1x + \cdots + a_{d-1}x^{d-1}
    \end{equation*}
    In some cases, especially with high degree polynomials, it can be useful to split the polynomials into smaller segments, which can be easier to handle. 
    The standard \textit{Kronecker segmentation} for multiplying binary polynomials is performed as follows
    \begin{enumerate}
        \item Split the polynomials into $n = \lceil d/w \rceil$ $w$-bit blocks:
            \begin{align*}
                a(x) &= a_0 + a_1x + \cdots + a_{d-1}x^{d-1} \\
                &= (a_0 + \cdots + a_{w-1}x^{w-1}) + \cdots + (a_{d-w} + \cdots + a_{d-1}x^{w-1})x^{w(n-1)}
            \end{align*}

        \item Define the field $\mathds{F}_{2^{2w}}:=\mathds{F}_2[z]/g(z)$, with $g(z)$ irreducible polynomial of degree $2w$.
        \item Define the map $\psi: \mathds{F}_2[x] \rightarrow \mathds{F}_{2^{2w}[y]}$ such that:
        \begin{equation*}
            a^{\prime}(y) = \psi(a(x)) := a_0^{\prime} + a^{\prime}_1y + \cdots + a^{\prime}_{n-1}y^{n-1}
        \end{equation*}
        where $a^{\prime}_i = (a_{iw} + a_{iw + 1}z + \cdots + a_{iw+(w-1)}z^{w-1})$.
    
        \item Calculate the product $c^{\prime}(y) = a^{\prime}(y)\cdot b^{\prime}(y) = c_0^{\prime} + c^{\prime}_1y + \cdots \in \mathds{F}_{2^{2w}}[y]$ \label{list:mult}
        \item Map $z \mapsto x$ and $y \mapsto x^w$, collect terms and coefficients and find the result.
    \end{enumerate}
    At step~\eqref{list:mult} we employ the FFT-based method to perform the pointwise multiplication.
